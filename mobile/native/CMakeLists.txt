cmake_minimum_required(VERSION 3.18)
project(lume_room LANGUAGES C CXX)
add_library(lume SHARED room.cpp peer_exchange.cpp vendor/monocypher.c vendor/utf8proc/utf8proc.c decoder.cpp local_media.cpp)
set(player_assets "${CMAKE_CURRENT_SOURCE_DIR}/../player-assets/player")
set(guest_files "${CMAKE_CURRENT_SOURCE_DIR}/guest.html" "${CMAKE_CURRENT_SOURCE_DIR}/guest.js" "${player_assets}/outputs.js" "${player_assets}/studio-theme.css")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${guest_files})
file(READ "${player_assets}/studio-theme.css" studio_theme)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/guest.html" guest_template)
string(CONFIGURE "${guest_template}" guest_html @ONLY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/guest.js" guest_js)
file(READ "${player_assets}/outputs.js" output_js)
configure_file(guest_assets.hpp.in guest_assets.hpp @ONLY)
target_include_directories(lume PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(lume PRIVATE cxx_std_17)
target_compile_options(lume PRIVATE -O2 -fvisibility=hidden)
if(ANDROID)
    target_sources(lume PRIVATE android.cpp)
    target_link_libraries(lume PRIVATE log android)
else()
    target_sources(lume PRIVATE harmony.cpp)
    target_link_libraries(lume PRIVATE libace_napi.z.so libohaudio.so)
endif()
